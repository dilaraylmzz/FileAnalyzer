using System.Collections.Generic;
using System.Text;
using UglyToad.PdfPig;

namespace FileAnalyzer.Readers
{
    public class PdfFileReader : IFileReader
    {
        public IReadOnlyCollection<string> SupportedExtensions { get; } = new[] { ".pdf" };

        public string Read(string filePath)
        {
            try
            {
                var builder = new StringBuilder();
                using (var document = PdfDocument.Open(filePath))
                {
                    foreach (var page in document.GetPages())
                        builder.AppendLine(page.Text);
                }
                return builder.ToString();
            }
            catch (UglyToad.PdfPig.Core.PdfDocumentFormatException ex)
            {
                throw new FileReadException("Dosya bozuk veya geçerli bir .pdf dosyası değil.", ex);
            }
        }
    }
}
